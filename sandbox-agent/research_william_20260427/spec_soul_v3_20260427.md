### 4.2 Skill Sandbox Isolation (ALICE Falta1 — SECURITY CRÍTICO)

**Problema:** Skills como código ejecutable sin aislamiento = backdoor para fallos catastróficos (rm -rf, exfiltración, etc.).

**Solution:** Three execution environments via Docker:
- `sandbox` (default): Docker container con read-only filesystem, no network, capabilities dropped (no sudo, no mount, no chmod). Whitelist explícita de paths writable (e.g., `/tmp/skill_workspace`).
- `restricted`: Read-only paths fuera de skill workspace + network whitelist específica (Soul DB, webchat). Para skills validadas (success_count ≥10).
- `unrestricted`: Como agente normal. **Requiere William approval explícito.** Audit log obligatorio.

```python
async def skill_execute(skill_id: int, params: dict):
    skill = await fetchrow("SELECT * FROM skills WHERE id=$1", skill_id)
    env_config = SANDBOX_CONFIG[skill.execution_environment]
    
    if skill.execution_environment == 'unrestricted':
        if not await william_approval_check(skill_id):
            raise PermissionError("Unrestricted skill requires William approval")
    
    async with docker_sandbox(env_config) as container:
        return await container.run(skill.skill_path, params)
```

### 4.3 Voyager Iterative Prompting (ALICE P3 — explicit spec)

```python
async def skill_synthesize(agent: str, task_description: str, max_iter: int = 4):
    """
    Voyager iterative prompting:
    - Max 4 iteraciones por skill
    - 3 tipos de feedback: env (bot.chat-style), execution errors, self-verification
    - Self-verification = LLM critic SEPARADO (no auto-validación)
    """
    rate_limit_check(agent)  # JARVIS H3 — max N skill_synthesize per day per agent
    
    for i in range(max_iter):
        # 1. Retrieve top-5 relevant skills
        relevant = await skill_search(agent, task_description, top_k=5)
        
        # 2. Generate with prev_attempts + env_feedback + relevant_skills + errors
        code = await llm_generate_skill(task_description, relevant, prev_errors=...)
        
        # 3. Execute en sandbox NEXUS
        result, errors = await sandbox_execute(code)
        
        # 4. Self-verification con LLM critic separado (NO mismo agente)
        passed = await llm_critic_separate(code, result, task_description)
        
        if passed:
            return await skill_commit(agent, name=..., code=code, ...)
        
        # 5. JARVIS H3 — early termination si iter 1-2 muy mal
        if i < 2 and verification_score(result) < 0.3:
            break  # don't waste budget if catastrophically wrong
    
    return None
```

**Rate limit (JARVIS H3):** Max 5 skill_synthesize per day per agent. Configurable in `agent_config`.

### 5.1 Reflective Optimizer (GEPA pattern, batch 4am — ADA 1.5)

```python
# Daemon seal-reflective-optimizer-batch corre 4am Lima (post-sleep_gate)
async def reflective_optimizer_batch():
    failed_traces = await pool.fetch("""
        SELECT * FROM reasoning_traces 
        WHERE outcome_success = FALSE 
          AND created_at > NOW() - INTERVAL '24 hours'
          AND id NOT IN (SELECT trace_id FROM reflective_diagnoses)
    """)
    
    for trace in failed_traces:
        # Cache check — si trace similar ya diagnosticada con confidence alta
        similar = await pool.fetchrow("""
            SELECT * FROM reflective_diagnoses 
            WHERE confidence > 0.85
              AND embedding <=> $1 < 0.15
            LIMIT 1
        """, trace.embedding)
        if similar:
            continue  # reusar diagnóstico previo
        
        # GEPA pattern: LLM diagnostica con trace completo
        diagnosis = await llm_diagnose(trace=trace, related_skill=...)
        
        if diagnosis.confidence > 0.7:
            await meta_proposal_create(
                proposed_by=trace.agent + "-reflective",
                target=diagnosis.target_table,
                type='skill_modify',
                diff=diagnosis.suggested_fix,
                rationale=diagnosis.root_cause,
                trace_evidence=trace.id,
                confidence=diagnosis.confidence
            )
        
        await pool.execute("""
            INSERT INTO reflective_diagnoses (trace_id, diagnosis, confidence, embedding, status)
            VALUES ($1, $2, $3, $4, 'pending_review')
        """, trace.id, diagnosis, diagnosis.confidence, diagnosis.embedding)
```

### 5.4 Meta-Proposal Rollback Flow (ADA G2)

```python
async def meta_proposal_rollback(original_id: int, reason: str, detected_by: str):
    """Cuando regresión es detectada post-apply."""
    # 1. Verify original was applied
    original = await fetchrow("SELECT * FROM meta_proposals WHERE id=$1 AND status='applied'", original_id)
    if not original:
        raise ValueError("Cannot rollback unapplied proposal")
    
    # 2. Generate inverse diff
    inverse_diff = await llm_generate_inverse_diff(original.diff)
    
    # 3. Create rollback proposal
    rollback = await meta_proposal_create(
        proposed_by=detected_by,
        target_agent=original.target_agent,
        type='rollback',
        tier=original.tier,
        diff=inverse_diff,
        rationale=f"Rollback of #{original_id}: {reason}",
        rollback_proposal_id=original_id
    )
    
    # 4. Test in NEXUS sandbox before promote
    test_result = await nexus_sandbox_validate(rollback.id)
    if not test_result.passed:
        return {"rollback_failed": True, "reason": test_result.errors}
    
    # 5. Apply (Tier-aware: emergency rollback can skip William if regression severe)
    if reason == 'critical_regression':
        return await apply_emergency_rollback(rollback.id)
    return await standard_meta_proposal_flow(rollback.id)
```

### 7.2 DAR Routing (refinado por ALICE M1)

```python
async def diversity_aware_router(message: dict, target_agent: str) -> bool:
    last_N = await pool.fetch("""
        SELECT content FROM chat_messages 
        WHERE recipients @> ARRAY[$1]
        ORDER BY id DESC LIMIT 20
    """, target_agent)
    
    msg_embedding = await embed_bge_m3(message['content'])
    last_embeddings = [await embed_bge_m3(m.content) for m in last_N]
    
    # Hear Both Sides: max disagreement
    diversity = 1.0 - max(cosine_similarity(msg_embedding, e) for e in last_embeddings)
    
    threshold = await pool.fetchval("""
        SELECT (persona_axes->>'dar_threshold')::float 
        FROM agent_config WHERE agent=$1
    """, target_agent)
    
    return diversity > threshold
```

async def semantic_density_gate(content: str) -> tuple[bool, list]:
    """Filtra phatic chit-chat, ACKs redundantes. Extrae factual statements auto-contenidas."""
    units = await llm_extract_factual_units(content, sliding_window=20)

```python
async def online_synthesis(new_units: list, existing_memories: list):
    """Consolida fragmentos durante WRITE, no en retrieval. Ejemplo: 3 fragmentos separados → 1 entrada coherente."""
    similar = await find_similar_recent(new_units, existing_memories, threshold=0.85)
    if similar:
        # Merge en vez de duplicar
        return await llm_consolidate(similar, new_units)
    return new_units

CREATE TABLE ocean_drift_log (
    id BIGSERIAL PRIMARY KEY,
    agent VARCHAR(20) NOT NULL,
    delta_o DECIMAL(3,2) CHECK (ABS(delta_o) <= 0.02),
    delta_c DECIMAL(3,2) CHECK (ABS(delta_c) <= 0.02),
    delta_e DECIMAL(3,2) CHECK (ABS(delta_e) <= 0.02),
    delta_a DECIMAL(3,2) CHECK (ABS(delta_a) <= 0.02),
    delta_n DECIMAL(3,2) CHECK (ABS(delta_n) <= 0.02),
    cumulative_o DECIMAL(3,2) CHECK (ABS(cumulative_o) <= 0.30),
    cumulative_c DECIMAL(3,2) CHECK (ABS(cumulative_c) <= 0.30),
    cumulative_e DECIMAL(3,2) CHECK (ABS(cumulative_e) <= 0.30),
    cumulative_a DECIMAL(3,2) CHECK (ABS(cumulative_a) <= 0.30),
    cumulative_n DECIMAL(3,2) CHECK (ABS(cumulative_n) <= 0.30),
    triggered_by VARCHAR(64),                   -- evento que causó el drift
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Trigger para 90-day rolling window damping (ADA 1.6)
CREATE OR REPLACE FUNCTION ocean_drift_check_lifetime() RETURNS trigger AS $$
DECLARE lifetime_drift FLOAT;
BEGIN
    SELECT COALESCE(SUM(ABS(delta_o)), 0) INTO lifetime_drift
    FROM ocean_drift_log
    WHERE agent = NEW.agent AND created_at > NOW() - INTERVAL '90 days';
    
    IF lifetime_drift + ABS(NEW.delta_o) > 0.30 THEN
        NEW.delta_o := NEW.delta_o * 0.5;  -- damping
    END IF;
    -- (similar para C, E, A, N)
    RETURN NEW;
END $$ LANGUAGE plpgsql;
```
